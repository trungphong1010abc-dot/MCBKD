   70   bool Dht22Sensor::readRaw40Bits(uint8_t bytes[5])
   71   {
   72       pinMode(_pin, OUTPUT_OPEN_DRAIN);
   73       digitalWrite(_pin, LOW);
   74       delay(20);
   75       digitalWrite(_pin, HIGH);
   76       delayMicroseconds(40);
   77       pinMode(_pin, INPUT_PULLUP);
   78
   79       uint32_t duration = 0;
   80       if (!expectPulse(LOW, 100, duration) || !expectPulse(HIGH, 100, duration))
   81       {
   82           return false;
   83       }
   84
   85       for (uint8_t bitIndex = 0; bitIndex < 40; bitIndex++)
   86       {
   87           if (!expectPulse(LOW, 80, duration) || !expectPulse(HIGH, 100, duration))
   88           {
   89               return false;
   90           }
   91
   92           if (duration > 50)
   93           {
   94               bytes[bitIndex / 8] |= (1 << (7 - (bitIndex % 8)));
   95           }
   96       }
   97
   98       return true;
   99   }

  101   bool Dht22Sensor::expectPulse(uint8_t level, uint32_t timeoutUs, uint32_t &durationUs) const
  102   {
  103       const uint32_t start = micros();

  105       {

  119       }
